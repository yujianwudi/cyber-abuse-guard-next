func TestCyberAbuseGuardHostFailOpenFixtureMatrix(t *testing.T) {
	t.Run("guard-rpc-registers-schema2-request-lifecycle", func(t *testing.T) {
		plugin := contractSchema2RequestLifecyclePlugin("cyber-abuse-guard", func(_ context.Context, _ pluginapi.RequestInterceptRequest) (pluginapi.RequestInterceptResponse, error) {
			return contractTermination("cyber-abuse-guard"), nil
		}, nil)
		plugin.Capabilities.ModelRouter = modelRouterFunc(func(_ context.Context, request pluginapi.ModelRouteRequest) (pluginapi.ModelRouteResponse, error) {
			if request.SourceFormat == "codex-alpha-search" {
				return pluginapi.ModelRouteResponse{Handled: true, TargetKind: pluginapi.ModelRouteTargetSelf}, nil
			}
			return pluginapi.ModelRouteResponse{Handled: false}, nil
		})
		lookup := newTestSymbolLookup(&testPlugin{registerResult: plugin})

		registered, errRegister := registerRPCPlugin(context.Background(), nil, "cyber-abuse-guard", lookup, pluginabi.MethodPluginRegister, []byte("mode: enforce"))
		if errRegister != nil {
			t.Fatalf("register schema-2 Guard: %v", errRegister)
		}
		if pluginabi.ABIVersion != 1 || pluginabi.SchemaVersion != 2 || lookup.lastLifecycle.SchemaVersion != 2 {
			t.Fatalf("Guard registration versions = ABI %d host schema %d lifecycle schema %d, want 1/2/2",
				pluginabi.ABIVersion, pluginabi.SchemaVersion, lookup.lastLifecycle.SchemaVersion)
		}
		if registered.Capabilities.RequestInterceptor == nil || registered.Capabilities.RequestLifecyclePlugin == nil {
			t.Fatalf("schema-2 Guard capabilities = %#v, want request interceptor and lifecycle plugin", registered.Capabilities)
		}
		if registered.Capabilities.ModelRouter == nil {
			t.Fatal("schema-2 Guard omitted the Alpha Search compatibility ModelRouter")
		}
		ordinary, errOrdinary := registered.Capabilities.ModelRouter.RouteModel(context.Background(), pluginapi.ModelRouteRequest{SourceFormat: "openai"})
		if errOrdinary != nil || ordinary.Handled {
			t.Fatalf("ordinary schema-2 Guard ModelRouter callback = %#v, %v; want unhandled", ordinary, errOrdinary)
		}
		alpha, errAlpha := registered.Capabilities.ModelRouter.RouteModel(context.Background(), pluginapi.ModelRouteRequest{SourceFormat: "codex-alpha-search"})
		if errAlpha != nil || !alpha.Handled || alpha.TargetKind != pluginapi.ModelRouteTargetSelf {
			t.Fatalf("Alpha Search compatibility ModelRouter callback = %#v, %v", alpha, errAlpha)
		}
	})

	t.Run("guard-priority-high-terminates-first", func(t *testing.T) {
		otherCalled := false
		host := newHostWithRecords(
			contractFallbackRecord("zzz-interceptor", 100, func() { otherCalled = true }),
			contractTerminatingRecord("cyber-abuse-guard", 300, nil),
		)
		assertContractTermination(t, host, "cyber-abuse-guard")
		if otherCalled {
			t.Fatal("lower-priority interceptor ran after Guard termination")
		}
	})

	t.Run("other-priority-high-terminates-before-guard", func(t *testing.T) {
		guardCalled := false
		host := newHostWithRecords(
			contractTerminatingRecord("cyber-abuse-guard", 300, func() { guardCalled = true }),
			contractTerminatingRecord("zzz-interceptor", 400, nil),
		)
		assertContractTermination(t, host, "zzz-interceptor")
		if guardCalled {
			t.Fatal("Guard ran after a higher-priority interceptor terminated the request")
		}
	})

	t.Run("same-priority-uses-plugin-id-ascending", func(t *testing.T) {
		hostGuardFirst := newHostWithRecords(
			contractTerminatingRecord("zzz-interceptor", 300, nil),
			contractTerminatingRecord("cyber-abuse-guard", 300, nil),
		)
		assertContractTermination(t, hostGuardFirst, "cyber-abuse-guard")

		hostFixtureFirst := newHostWithRecords(
			contractTerminatingRecord("cyber-abuse-guard", 300, nil),
			contractTerminatingRecord("aaa-interceptor", 300, nil),
		)
		assertContractTermination(t, hostFixtureFirst, "aaa-interceptor")
	})

	t.Run("guard-fused-falls-through", func(t *testing.T) {
		host := newHostWithRecords(
			contractFallbackRecord("zzz-interceptor", 100, nil),
			contractTerminatingRecord("cyber-abuse-guard", 300, nil),
		)
		host.fusePlugin("cyber-abuse-guard", "fixture", "forced fuse")
		assertContractFallback(t, host)
	})

	t.Run("guard-interceptor-error-falls-through", func(t *testing.T) {
		host := newHostWithRecords(
			contractFallbackRecord("zzz-interceptor", 100, nil),
			contractRequestLifecycleRecord("cyber-abuse-guard", 300, func(context.Context, pluginapi.RequestInterceptRequest) (pluginapi.RequestInterceptResponse, error) {
				return pluginapi.RequestInterceptResponse{}, errors.New("forced Guard interceptor error")
			}, nil),
		)
		assertContractFallback(t, host)
	})

	t.Run("guard-interceptor-panic-fuses-and-falls-through", func(t *testing.T) {
		host := newHostWithRecords(
			contractFallbackRecord("zzz-interceptor", 100, nil),
			contractRequestLifecycleRecord("cyber-abuse-guard", 300, func(context.Context, pluginapi.RequestInterceptRequest) (pluginapi.RequestInterceptResponse, error) {
				panic("forced Guard interceptor panic")
			}, nil),
		)
		assertContractFallback(t, host)
		if !host.isPluginFused("cyber-abuse-guard") {
			t.Fatal("Guard interceptor panic did not fuse the plugin")
		}
	})

	t.Run("guard-failure-without-fallback-continues-native", func(t *testing.T) {
		host := newHostWithRecords(contractRequestLifecycleRecord(
			"cyber-abuse-guard",
			300,
			func(context.Context, pluginapi.RequestInterceptRequest) (pluginapi.RequestInterceptResponse, error) {
				return pluginapi.RequestInterceptResponse{}, errors.New("forced Guard interceptor error")
			},
			nil,
		))
		response := host.InterceptRequestBeforeAuth(context.Background(), contractInterceptRequest())
		if response.Terminate || string(response.Body) != `{"model":"fixture-model"}` || response.Headers.Get("X-Input") != "preserve" {
			t.Fatalf("failed Guard interceptor response = %#v, want unchanged native request", response)
		}
	})

	t.Run("request-interceptor-metadata-sanitizer-drops-non-json-callback", func(t *testing.T) {
		request := rpcRequestInterceptRequest{
			RequestInterceptRequest: pluginapi.RequestInterceptRequest{Metadata: map[string]any{
				"keep": "value",
				coreexecutor.SelectedAuthIndexCallbackMetadataKey: func(string) {},
			}},
			HostCallbackID: "callback-1",
		}
		sanitized, ok := sanitizePluginRequest(request).(rpcRequestInterceptRequest)
		if !ok {
			t.Fatalf("sanitizePluginRequest() type = %T, want rpcRequestInterceptRequest", sanitizePluginRequest(request))
		}
		raw, errMarshal := json.Marshal(sanitized)
		if errMarshal != nil {
			t.Fatalf("marshal sanitized RequestInterceptor request: %v", errMarshal)
		}
		var decoded rpcRequestInterceptRequest
		if errUnmarshal := json.Unmarshal(raw, &decoded); errUnmarshal != nil {
			t.Fatalf("unmarshal sanitized RequestInterceptor request: %v", errUnmarshal)
		}
		if decoded.Metadata["keep"] != "value" {
			t.Fatalf("sanitized metadata = %#v, want preserved keep value", decoded.Metadata)
		}
		if _, exists := decoded.Metadata[coreexecutor.SelectedAuthIndexCallbackMetadataKey]; exists {
			t.Fatalf("selected-auth-index callback crossed the C-ABI JSON boundary: %#v", decoded.Metadata)
		}
		if decoded.HostCallbackID != "callback-1" {
			t.Fatalf("host callback ID = %q, want callback-1", decoded.HostCallbackID)
		}
	})

	t.Run("request-lifecycle-reports-all-schema2-outcomes", func(t *testing.T) {
		completions := make(chan pluginapi.RequestCompletion, 1)
		host := newHostWithRecords(contractRequestLifecycleRecord(
			"cyber-abuse-guard",
			300,
			func(_ context.Context, req pluginapi.RequestInterceptRequest) (pluginapi.RequestInterceptResponse, error) {
				return pluginapi.RequestInterceptResponse{Headers: req.Headers, Body: req.Body}, nil
			},
			func(_ context.Context, completion pluginapi.RequestCompletion) { completions <- completion },
		))
		for _, outcome := range []pluginapi.RequestCompletionOutcome{
			pluginapi.RequestCompletionSucceeded,
			pluginapi.RequestCompletionFailed,
			pluginapi.RequestCompletionRejected,
			pluginapi.RequestCompletionCanceled,
		} {
			requestID := "completion-" + string(outcome)
			host.CompleteRequest(context.Background(), pluginapi.RequestCompletion{RequestID: requestID, Outcome: outcome})
			select {
			case got := <-completions:
				if got.RequestID != requestID || got.Outcome != outcome {
					t.Fatalf("completion = %#v, want request %q outcome %q", got, requestID, outcome)
				}
			case <-time.After(time.Second):
				t.Fatalf("timed out waiting for lifecycle outcome %q", outcome)
			}
		}
	})

	for _, lifecycle := range []string{"guard-not-loaded", "guard-registration-failed", "guard-disabled"} {
		t.Run(lifecycle+"-falls-through", func(t *testing.T) {
			assertLifecycleFallback(t, lifecycle)
		})
	}

	t.Run("legacy-schema1-model-router-compatibility-fixture", func(t *testing.T) {
		plugin := validTestPlugin("legacy-router-schema1")
		plugin.Capabilities.ModelRouter = modelRouterFunc(func(context.Context, pluginapi.ModelRouteRequest) (pluginapi.ModelRouteResponse, error) {
			return pluginapi.ModelRouteResponse{Handled: true, TargetKind: pluginapi.ModelRouteTargetProvider, Target: "legacy-provider"}, nil
		})
		lookup := newTestSymbolLookup(&testPlugin{registerResult: plugin})
		lookup.schemaVersion = 1

		registered, errRegister := registerRPCPlugin(context.Background(), nil, "legacy-router-schema1", lookup, pluginabi.MethodPluginRegister, nil)
		if errRegister != nil {
			t.Fatalf("register explicit schema-1 compatibility Router: %v", errRegister)
		}
		if registered.Capabilities.ModelRouter == nil {
			t.Fatal("schema-1 compatibility ModelRouter = nil")
		}
		if registered.Capabilities.RequestInterceptor != nil || registered.Capabilities.RequestLifecyclePlugin != nil {
			t.Fatal("schema-1 compatibility Router unexpectedly exposes schema-2 request lifecycle capabilities")
		}
		response, errRoute := registered.Capabilities.ModelRouter.RouteModel(context.Background(), pluginapi.ModelRouteRequest{RequestedModel: "legacy-model"})
		if errRoute != nil || !response.Handled || response.Target != "legacy-provider" {
			t.Fatalf("schema-1 compatibility RouteModel() = %#v, %v", response, errRoute)
		}
	})
}

func contractTermination(id string) pluginapi.RequestInterceptResponse {
	return pluginapi.RequestInterceptResponse{
		Terminate:       true,
		StatusCode:      http.StatusUnavailableForLegalReasons,
		ResponseHeaders: http.Header{"Content-Type": []string{"application/json"}, "X-Cag-Terminated-By": []string{id}},
		ResponseBody:    []byte(`{"error":"blocked"}`),
	}
}

func contractTerminatingRecord(id string, priority int, called func()) capabilityRecord {
	return contractRequestLifecycleRecord(id, priority, func(context.Context, pluginapi.RequestInterceptRequest) (pluginapi.RequestInterceptResponse, error) {
		if called != nil {
			called()
		}
		return contractTermination(id), nil
	}, nil)
}

func contractFallbackRecord(id string, priority int, called func()) capabilityRecord {
	return contractRequestLifecycleRecord(id, priority, func(_ context.Context, req pluginapi.RequestInterceptRequest) (pluginapi.RequestInterceptResponse, error) {
		if called != nil {
			called()
		}
		return pluginapi.RequestInterceptResponse{
			Headers: http.Header{"X-Fixture-Fallback": []string{id}},
			Body:    append(req.Body, []byte("|fallback")...),
		}, nil
	}, nil)
}

func contractRequestLifecycleRecord(
	id string,
	priority int,
	intercept func(context.Context, pluginapi.RequestInterceptRequest) (pluginapi.RequestInterceptResponse, error),
	complete func(context.Context, pluginapi.RequestCompletion),
) capabilityRecord {
	return capabilityRecord{
		id:       id,
		priority: priority,
		plugin:   contractSchema2RequestLifecyclePlugin(id, intercept, complete),
	}
}

func contractSchema2RequestLifecyclePlugin(
	name string,
	intercept func(context.Context, pluginapi.RequestInterceptRequest) (pluginapi.RequestInterceptResponse, error),
	complete func(context.Context, pluginapi.RequestCompletion),
) pluginapi.Plugin {
	plugin := validTestPlugin(name)
	plugin.Capabilities.RequestInterceptor = requestInterceptorFunc(intercept)
	plugin.Capabilities.RequestLifecyclePlugin = requestLifecyclePluginFunc(func(ctx context.Context, completion pluginapi.RequestCompletion) {
		if complete != nil {
			complete(ctx, completion)
		}
	})
	return plugin
}

func contractInterceptRequest() pluginapi.RequestInterceptRequest {
	return pluginapi.RequestInterceptRequest{
		RequestID:      "fixture-request",
		SourceFormat:   "openai",
		Model:          "fixture-model",
		RequestedModel: "fixture-model",
		Headers:        http.Header{"X-Input": []string{"preserve"}},
		Body:           []byte(`{"model":"fixture-model"}`),
	}
}

func assertContractTermination(t *testing.T, host *Host, want string) {
	t.Helper()
	response := host.InterceptRequestBeforeAuth(context.Background(), contractInterceptRequest())
	if !response.Terminate || response.StatusCode != http.StatusUnavailableForLegalReasons ||
		response.ResponseHeaders.Get("X-CAG-Terminated-By") != want || string(response.ResponseBody) != `{"error":"blocked"}` {
		t.Fatalf("InterceptRequestBeforeAuth() = %#v, want termination by %q", response, want)
	}
}

func assertContractFallback(t *testing.T, host *Host) {
	t.Helper()
	response := host.InterceptRequestBeforeAuth(context.Background(), contractInterceptRequest())
	if response.Terminate || response.Headers.Get("X-Fixture-Fallback") != "zzz-interceptor" ||
		string(response.Body) != `{"model":"fixture-model"}|fallback` {
		t.Fatalf("InterceptRequestBeforeAuth() = %#v, want fixture fallthrough", response)
	}
}

func assertLifecycleFallback(t *testing.T, scenario string) {
	t.Helper()
	enabled := true
	disabled := false
	loader := newTestSymbolLoader()
	fixturePlugin := contractSchema2RequestLifecyclePlugin("fixture", func(_ context.Context, req pluginapi.RequestInterceptRequest) (pluginapi.RequestInterceptResponse, error) {
		return pluginapi.RequestInterceptResponse{
			Headers: http.Header{"X-Fixture-Fallback": []string{"zzz-interceptor"}},
			Body:    append(req.Body, []byte("|fallback")...),
		}, nil
	}, nil)
	fixture := &testPlugin{registerResult: fixturePlugin, reconfigureResult: fixturePlugin}
	fixtureLookup := newTestSymbolLookup(fixture)
	loader.lookups["zzz-interceptor"] = fixtureLookup

	guardPlugin := contractSchema2RequestLifecyclePlugin("cyber-abuse-guard", func(context.Context, pluginapi.RequestInterceptRequest) (pluginapi.RequestInterceptResponse, error) {
		return contractTermination("cyber-abuse-guard"), nil
	}, nil)
	guard := &testPlugin{registerResult: guardPlugin, reconfigureResult: guardPlugin}
	guardConfig := config.PluginInstanceConfig{Enabled: &enabled, Priority: 300}
	switch scenario {
	case "guard-not-loaded":
		// Leave the loader entry absent while discovery still finds the file.
	case "guard-registration-failed":
		guard.panicOnRegister = true
		loader.lookups["cyber-abuse-guard"] = newTestSymbolLookup(guard)
	case "guard-disabled":
		guardConfig.Enabled = &disabled
		loader.lookups["cyber-abuse-guard"] = newTestSymbolLookup(guard)
	default:
		t.Fatalf("unknown lifecycle scenario %q", scenario)
	}

	host := NewForTest(loader)
	t.Cleanup(host.ShutdownAll)
	host.ApplyConfig(context.Background(), &config.Config{Plugins: config.PluginsConfig{
		Enabled: true,
		Dir:     makePluginDir(t, "cyber-abuse-guard", "zzz-interceptor"),
		Configs: map[string]config.PluginInstanceConfig{
			"cyber-abuse-guard": guardConfig,
			"zzz-interceptor":   {Enabled: &enabled, Priority: 100},
		},
	}})

	assertContractFallback(t, host)
	if fixtureLookup.lastLifecycle.SchemaVersion != 2 {
		t.Fatalf("fallback interceptor lifecycle schema_version = %d, want 2", fixtureLookup.lastLifecycle.SchemaVersion)
	}
	if host.PluginRegistered("cyber-abuse-guard") {
		t.Fatalf("Guard unexpectedly registered in scenario %q", scenario)
	}
}
