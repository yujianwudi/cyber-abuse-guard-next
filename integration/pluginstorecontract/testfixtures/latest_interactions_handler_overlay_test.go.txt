func TestCyberAbuseGuardInteractionsTranslatorRegistryContract(t *testing.T) {
	if sdktranslator.FormatInteractions.String() != "interactions" {
		t.Fatalf("FormatInteractions = %q, want interactions", sdktranslator.FormatInteractions)
	}
	if !sdktranslator.HasRequestTransformer(sdktranslator.FormatInteractions, sdktranslator.FormatOpenAI) {
		t.Fatal("latest CPA translator registry has no interactions -> openai request transformer")
	}
	if !sdktranslator.HasNonStreamResponseTransformer(sdktranslator.FormatInteractions, sdktranslator.FormatOpenAI) {
		t.Fatal("latest CPA translator registry has no openai -> interactions non-stream response transformer")
	}
	if !sdktranslator.HasStreamResponseTransformer(sdktranslator.FormatInteractions, sdktranslator.FormatOpenAI) {
		t.Fatal("latest CPA translator registry has no openai -> interactions stream response transformer")
	}
}

func TestCyberAbuseGuardInteractionsHandlerRequestInterceptorContract(t *testing.T) {
	if pluginabi.ABIVersion != 1 || pluginabi.SchemaVersion != 2 {
		t.Fatalf("CPA plugin versions = ABI %d schema %d, want 1/2", pluginabi.ABIVersion, pluginabi.SchemaVersion)
	}
	gin.SetMode(gin.TestMode)
	const registryClientID = "cyber-abuse-guard-interactions-schema2-contract"
	registry.GetGlobalRegistry().RegisterClient(registryClientID, "gemini", []*registry.ModelInfo{{ID: "gemini-3.5-flash"}})
	t.Cleanup(func() {
		registry.GetGlobalRegistry().UnregisterClient(registryClientID)
	})

	const inputCanary = "INTERACTIONS_REQUEST_INTERCEPTOR_CANARY"
	body := `{"model":"models/gemini-3.5-flash","input":[{"type":"user_input","content":[{"type":"text","text":"` + inputCanary + `"}]}]}`
	host := &interactionsRequestLifecycleContractHost{}
	recorder := runInteractionsContractRequest(t, host, body, "schema2-session")

	if recorder.Code != http.StatusUnavailableForLegalReasons {
		t.Fatalf("status = %d, want %d; body=%s", recorder.Code, http.StatusUnavailableForLegalReasons, recorder.Body.String())
	}
	if recorder.Header().Get("X-CAG-Policy") != "blocked" || recorder.Body.String() != `{"blocked":true}` {
		t.Fatalf("termination response headers/body = %#v / %s", recorder.Header(), recorder.Body.String())
	}
	if host.beforeCalls != 1 || host.afterCalls != 0 || host.responseCalls != 0 || host.streamCalls != 0 {
		t.Fatalf("interceptor calls before/after/response/stream = %d/%d/%d/%d, want 1/0/0/0",
			host.beforeCalls, host.afterCalls, host.responseCalls, host.streamCalls)
	}
	request := host.request
	if request.RequestID == "" || request.SourceFormat != sdktranslator.FormatInteractions.String() ||
		request.Model != "gemini-3.5-flash" || request.RequestedModel != "gemini-3.5-flash" || request.Stream || request.ToFormat != "" {
		t.Fatalf("request identity = %#v", request)
	}
	if request.Headers.Get("X-CAG-Visibility") != "header-canary" {
		t.Fatalf("request headers = %#v, want visibility canary", request.Headers)
	}
	if got := gjson.GetBytes(request.Body, "model").String(); got != "gemini-3.5-flash" {
		t.Fatalf("intercepted body model = %q, want normalized gemini-3.5-flash; body=%s", got, request.Body)
	}
	if got := gjson.GetBytes(request.Body, "input.0.content.0.text").String(); got != inputCanary {
		t.Fatalf("intercepted body input = %q, want %q; body=%s", got, inputCanary, request.Body)
	}
	completion := host.completion
	if host.completionCalls != 1 || completion.RequestID != request.RequestID ||
		completion.SourceFormat != sdktranslator.FormatInteractions.String() ||
		completion.Model != "gemini-3.5-flash" || completion.RequestedModel != "gemini-3.5-flash" ||
		completion.Outcome != pluginapi.RequestCompletionRejected || completion.StatusCode != http.StatusUnavailableForLegalReasons {
		t.Fatalf("completion calls=%d completion=%#v", host.completionCalls, completion)
	}
	if completion.StartedAt.IsZero() || completion.CompletedAt.Before(completion.StartedAt) {
		t.Fatalf("completion timestamps = %#v", completion)
	}
}

type interactionsRequestLifecycleContractHost struct {
	beforeCalls     int
	afterCalls      int
	responseCalls   int
	streamCalls     int
	completionCalls int
	request         pluginapi.RequestInterceptRequest
	completion      pluginapi.RequestCompletion
}

func (h *interactionsRequestLifecycleContractHost) InterceptRequestBeforeAuth(_ context.Context, request pluginapi.RequestInterceptRequest) pluginapi.RequestInterceptResponse {
	h.beforeCalls++
	h.request = request
	return pluginapi.RequestInterceptResponse{
		Terminate:       true,
		StatusCode:      http.StatusUnavailableForLegalReasons,
		ResponseHeaders: http.Header{"Content-Type": []string{"application/json"}, "X-Cag-Policy": []string{"blocked"}},
		ResponseBody:    []byte(`{"blocked":true}`),
	}
}

func (h *interactionsRequestLifecycleContractHost) InterceptRequestAfterAuth(_ context.Context, request pluginapi.RequestInterceptRequest) pluginapi.RequestInterceptResponse {
	h.afterCalls++
	return pluginapi.RequestInterceptResponse{Headers: request.Headers, Body: request.Body}
}

func (h *interactionsRequestLifecycleContractHost) InterceptResponse(_ context.Context, request pluginapi.ResponseInterceptRequest) pluginapi.ResponseInterceptResponse {
	h.responseCalls++
	return pluginapi.ResponseInterceptResponse{Headers: request.ResponseHeaders, Body: request.Body}
}

func (h *interactionsRequestLifecycleContractHost) InterceptStreamChunk(_ context.Context, request pluginapi.StreamChunkInterceptRequest) pluginapi.StreamChunkInterceptResponse {
	h.streamCalls++
	return pluginapi.StreamChunkInterceptResponse{Headers: request.ResponseHeaders, Body: request.Body}
}

func (h *interactionsRequestLifecycleContractHost) CompleteRequest(_ context.Context, completion pluginapi.RequestCompletion) {
	h.completionCalls++
	h.completion = completion
}

func runInteractionsContractRequest(t *testing.T, host *interactionsRequestLifecycleContractHost, body, session string) *httptest.ResponseRecorder {
	t.Helper()
	recorder := httptest.NewRecorder()
	ginContext, _ := gin.CreateTestContext(recorder)
	ginContext.Request = httptest.NewRequest(http.MethodPost, "/v1beta/interactions?session="+session, strings.NewReader(body))
	ginContext.Request.Header.Set("Content-Type", "application/json")
	ginContext.Request.Header.Set("X-CAG-Visibility", "header-canary")

	base := handlers.NewBaseAPIHandlers(&sdkconfig.SDKConfig{}, nil)
	base.SetPluginHost(host)
	NewGeminiAPIHandler(base).Interactions(ginContext)
	return recorder
}
