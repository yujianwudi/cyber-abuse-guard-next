package pluginhost

import (
	"bytes"
	"context"
	"net/http"
	"testing"
	"time"

	"github.com/router-for-me/CLIProxyAPI/v7/sdk/pluginabi"
	"github.com/router-for-me/CLIProxyAPI/v7/sdk/pluginapi"
	sdktranslator "github.com/router-for-me/CLIProxyAPI/v7/sdk/translator"
)

func TestCyberAbuseGuardInteractionsRequestLifecycleFormatContract(t *testing.T) {
	if pluginabi.ABIVersion != 1 || pluginabi.SchemaVersion != 2 {
		t.Fatalf("CPA plugin versions = ABI %d schema %d, want 1/2", pluginabi.ABIVersion, pluginabi.SchemaVersion)
	}

	body := []byte(`{"model":"gemini-3.5-flash","input":"INTERACTIONS_SCHEMA2_CANARY"}`)
	headers := make(http.Header)
	headers.Set("X-CAG-Visibility", "header-canary")
	completions := make(chan pluginapi.RequestCompletion, 1)

	plugin := validTestPlugin("cyber-abuse-guard")
	plugin.Capabilities.ModelRouter = modelRouterFunc(func(_ context.Context, request pluginapi.ModelRouteRequest) (pluginapi.ModelRouteResponse, error) {
		if request.SourceFormat == "codex-alpha-search" {
			return pluginapi.ModelRouteResponse{Handled: true, TargetKind: pluginapi.ModelRouteTargetSelf}, nil
		}
		return pluginapi.ModelRouteResponse{Handled: false}, nil
	})
	plugin.Capabilities.RequestInterceptor = requestInterceptorFunc(func(_ context.Context, request pluginapi.RequestInterceptRequest) (pluginapi.RequestInterceptResponse, error) {
		if request.RequestID != "interactions-schema2-request" || request.SourceFormat != sdktranslator.FormatInteractions.String() ||
			request.Model != "gemini-3.5-flash" || request.RequestedModel != "gemini-3.5-flash" || request.Stream {
			t.Fatalf("request identity = id %q format %q model %q requested %q stream %v",
				request.RequestID, request.SourceFormat, request.Model, request.RequestedModel, request.Stream)
		}
		if !bytes.Equal(request.Body, body) || request.Headers.Get("X-CAG-Visibility") != "header-canary" {
			t.Fatalf("request body/headers drifted: body=%s headers=%#v", request.Body, request.Headers)
		}
		return pluginapi.RequestInterceptResponse{
			Terminate:       true,
			StatusCode:      http.StatusUnavailableForLegalReasons,
			ResponseHeaders: http.Header{"Content-Type": []string{"application/json"}, "X-Cag-Policy": []string{"blocked"}},
			ResponseBody:    []byte(`{"blocked":true}`),
		}, nil
	})
	plugin.Capabilities.RequestLifecyclePlugin = requestLifecyclePluginFunc(func(_ context.Context, completion pluginapi.RequestCompletion) {
		completions <- completion
	})
	lookup := newTestSymbolLookup(&testPlugin{registerResult: plugin})

	registered, errRegister := registerRPCPlugin(context.Background(), nil, "cyber-abuse-guard", lookup, pluginabi.MethodPluginRegister, []byte("mode: enforce"))
	if errRegister != nil {
		t.Fatalf("register schema-2 interactions Guard: %v", errRegister)
	}
	if lookup.lastLifecycle.SchemaVersion != 2 {
		t.Fatalf("Guard lifecycle schema_version = %d, want 2", lookup.lastLifecycle.SchemaVersion)
	}
	if registered.Capabilities.RequestInterceptor == nil || registered.Capabilities.RequestLifecyclePlugin == nil {
		t.Fatalf("registered Guard capabilities = %#v, want request_interceptor and request_lifecycle_plugin", registered.Capabilities)
	}
	if registered.Capabilities.ModelRouter == nil {
		t.Fatal("current schema-2 Guard fixture omitted the Alpha Search compatibility ModelRouter")
	}

	host := newHostWithRecords(capabilityRecord{
		id:       "cyber-abuse-guard",
		priority: 300,
		plugin:   registered,
	})
	response := host.InterceptRequestBeforeAuth(context.Background(), pluginapi.RequestInterceptRequest{
		RequestID:      "interactions-schema2-request",
		SourceFormat:   sdktranslator.FormatInteractions.String(),
		Model:          "gemini-3.5-flash",
		RequestedModel: "gemini-3.5-flash",
		Headers:        headers,
		Body:           body,
	})
	if !response.Terminate || response.StatusCode != http.StatusUnavailableForLegalReasons ||
		response.ResponseHeaders.Get("X-CAG-Policy") != "blocked" || string(response.ResponseBody) != `{"blocked":true}` {
		t.Fatalf("schema-2 Guard termination response = %#v", response)
	}

	host.CompleteRequest(context.Background(), pluginapi.RequestCompletion{
		RequestID:      "interactions-schema2-request",
		SourceFormat:   sdktranslator.FormatInteractions.String(),
		Model:          "gemini-3.5-flash",
		RequestedModel: "gemini-3.5-flash",
		Outcome:        pluginapi.RequestCompletionRejected,
		StatusCode:     http.StatusUnavailableForLegalReasons,
	})
	select {
	case completion := <-completions:
		if completion.RequestID != "interactions-schema2-request" ||
			completion.SourceFormat != sdktranslator.FormatInteractions.String() ||
			completion.Outcome != pluginapi.RequestCompletionRejected ||
			completion.StatusCode != http.StatusUnavailableForLegalReasons {
			t.Fatalf("schema-2 Guard completion = %#v", completion)
		}
	case <-time.After(time.Second):
		t.Fatal("timed out waiting for schema-2 Guard completion")
	}
}
